```python
def _make_session(conn) -> int:
    """Create a minimal solve_session and return its id. Used as FK parent."""
    from database.models import SolveSessionCreate
    from database.crud import create_solve_session
    return create_solve_session(conn, SolveSessionCreate(
        selected_algorithm="Kociemba", status="pending"
    ))
```

```python
def _make_solution(conn, session_id: int) -> int:
    from database.models import SolutionCreate
    from database.crud import create_solution
    return create_solution(conn, SolutionCreate(
        session_id=session_id, algorithm_used="Kociemba",
        move_count=6, solution_string="U R2 F' B R U'",
        generated_by="rpi2-solver"
    ))

def _make_execution_run(conn, session_id: int, solution_id: int) -> int:
    from database.models import ExecutionRunCreate
    from database.crud import create_execution_run
    return create_execution_run(conn, ExecutionRunCreate(
        session_id=session_id, solution_id=solution_id,
        status="running", motor_node_id="rpi3-motors"
    ))
```
